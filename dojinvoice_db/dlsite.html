dojinvoice_db.dlsite

Request-efficient access layer over dlsite_async.

Three kinds of request are used, in increasing cost per work:

  1. Search listing (DlsiteClient.fetch_listing_ids()) — one request returns up to 100 product IDs, newest first.
  2. Product info ajax (DlsiteClient.fetch_product_info()) — DLsite's product/info/ajax endpoint accepts a comma-separated list of product IDs, so one request covers up to 100 works. It yields everything DlsiteAPI.product_info() yields (title, maker, work type, release date, cover image) plus the price/sales/rating figures.
  3. Work page HTML (DlsiteClient.fetch_detail()) — one request per work, the only source for circle name, description, genres and the voice actor / scenario / illustration / music credits.

Only step 3 scales with the number of works, so the crawler performs it once per work and never again unless explicitly asked to.

114def _to_datetime(value: Any) -> datetime | None:  # noqa: ANN401 - raw JSON value
115    if not isinstance(value, str):
116        return None
117    for fmt in ("%Y-%m-%d %H:%M:%S", "%Y-%m-%d"):
118        try:
119            return datetime.strptime(value, fmt)  # noqa: DTZ007 - DLsite times are naive JST
120        except ValueError:
121            continue
122    return None

125def work_from_info(product_id: str, info: Mapping[str, Any]) -> Work:
126    """Build a :class:`~dlsite_async.Work` from one ajax info entry.
127
128    Unlike ``DlsiteAPI.product_info()`` this never raises on unfamiliar enum
129    values -- DLsite occasionally introduces new work/book types, and dropping
130    the field is better than dropping the work.
131    """
132    data = dict(info)
133    data["product_id"] = product_id
134    data["age_category"] = _enum_or_none(AgeCategory, data.get("age_category"))
135    data["work_type"] = _enum_or_none(WorkType, data.get("work_type"))
136    book_type = data.get("book_type")
137    if isinstance(book_type, dict):
138        book_type = book_type.get("value")
139    data["book_type"] = _enum_or_none(BookType, book_type)
140    data["regist_date"] = _to_datetime(data.get("regist_date"))
141    return Work.from_dict(data)

145    if value is None:

147    try:

150        LOGGER.debug("unknown %s value: %r", enum_cls.__name__, value)
151        return None

154class _Throttle:
155    """Serialize requests so that at least ``interval`` seconds pass between them."""
156
157    def __init__(self, interval: float) -> None:
158        self._interval = max(interval, 0.0)
159        self._lock = asyncio.Lock()
160        self._next = 0.0
161
162    async def wait(self) -> None:
163        """Block until the next request is allowed to start."""
164        if not self._interval:
165            return
166        async with self._lock:
167            now = time.monotonic()
168            delay = self._next - now
169            if delay > 0:
170                await asyncio.sleep(delay)
171                now = time.monotonic()
172            self._next = now + self._interval

219    async def _get_text(self, url: str, *, params: Mapping[str, str] | None = None) -> str | None:
220        """GET ``url`` and return its body, or ``None`` on a 404/403.
221
222        Retries transient failures with exponential backoff.
223        """
224        last_error: Exception | None = None
225        for attempt in range(self._retries):
226            async with self._semaphore:
227                await self._throttle.wait()
228                self.request_count += 1
229                try:
230                    async with self.api.get(
231                        url,
232                        params=dict(params or {}),
233                        headers={"User-Agent": USER_AGENT},
234                        timeout=self._timeout,
235                    ) as response:
236                        return await response.text()
237                except ClientResponseError as e:
238                    if e.status not in _RETRY_STATUSES:
239                        if e.status in {403, 404}:
240                            return None
241                        raise
242                    last_error = e
243                except (ClientError, TimeoutError) as e:
244                    last_error = e
245            backoff = 2.0**attempt
246            LOGGER.warning("request failed (%s), retrying in %.0fs: %s", last_error, backoff, url)
247            await asyncio.sleep(backoff)
248        msg = f"giving up on {url}: {last_error}"
249        raise WorkNotFoundError(msg) from last_error

258    async def fetch_product_info(self, product_ids: Sequence[str]) -> dict[str, dict[str, Any]]:
259        """Return raw ajax info for up to ~100 product IDs in a single request.
260
261        IDs DLsite does not know about are simply absent from the result.
262        """
263        if not product_ids:
264            return {}

266        if not text:

271            LOGGER.warning("malformed info response for %d ids", len(product_ids))
272            return {}
273        if not isinstance(data, dict):
274            return {}
275        return {k: v for k, v in data.items() if isinstance(v, dict)}

277    async def fetch_detail(self, work: Work) -> Work:
278        """Fill in the work-page-only fields of ``work``.
279
280        Raises:
281            WorkNotFoundError: Neither the work nor the announce page exists.
282        """
283        if parse_work_html is None:  # pragma: no cover - dependency fallback
284            return await self.api.get_work(work.product_id)
285        site_id = work.site_id or "maniax"
286        for kind in ("work", "announce"):
287            url = f"https://www.dlsite.com/{site_id}/{kind}/=/product_id/{work.product_id}.html/"
288            html = await self._get_text(url)
289            if html:
290                return replace(work, **parse_detail_html(html))
291        msg = f"no work or announce page for {work.product_id}"
292        raise WorkNotFoundError(msg)

295def batched(items: Sequence[str], size: int) -> Iterable[Sequence[str]]:
296    """Split ``items`` into consecutive chunks of at most ``size``."""
297    step = max(size, 1)
298    for start in range(0, len(items), step):
299        yield items[start : start + step]
