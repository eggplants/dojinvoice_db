dojinvoice_db.db

SQLite storage for DLsite voice works.

The schema separates three kinds of data by how expensive they are to obtain and how often they change:

  • work — metadata. summary_fetched_at marks the cheap ajax-sourced columns, detail_fetched_at the ones that cost one request per work. A NULL detail_fetched_at is the resume marker: those works are picked up by the next run.
  • work_stat (+ work_stat_history) — price/sales/rating figures, which keep changing and can be refreshed in batches of 100 works per request.
  • work_creator / work_genre / work_file_format / work_event / work_sample_image — the multi-valued detail fields.

fetch_error remembers works that failed so repeated runs do not keep re-requesting them, and crawl_state holds bookkeeping between runs.

220def to_iso(value: datetime | None) -> str | None:
221    """Render a datetime the way it is stored (``YYYY-MM-DD HH:MM:SS``)."""
222    if value is None:
223        return None
224    return value.isoformat(sep=" ", timespec="seconds")

227def from_iso(value: str | None) -> datetime | None:
228    """Parse a datetime stored by :func:`to_iso`."""
229    if not value:
230        return None
231    try:
232        return datetime.fromisoformat(value)
233    except ValueError:  # pragma: no cover - defensive
234        return None

242    except ValueError:  # pragma: no cover - defensive
243        return None

289    def __init__(self, path: str | Path) -> None:
290        self.path = str(path)
291        if self.path != ":memory:":
292            Path(self.path).parent.mkdir(parents=True, exist_ok=True)
293        self.conn = sqlite3.connect(self.path)
294        self.conn.row_factory = sqlite3.Row
295        self.conn.execute("pragma journal_mode = wal")
296        self.conn.execute("pragma foreign_keys = on")
297        try:
298            self._migrate()
299        except Exception:
300            self.conn.close()
301            raise

320    def _migrate(self) -> None:
321        version = int(self.conn.execute("pragma user_version").fetchone()[0])
322        if version == 0:
323            with self.conn:
324                self.conn.executescript(_SCHEMA)
325                self.conn.execute(f"pragma user_version = {SCHEMA_VERSION}")
326        elif version > SCHEMA_VERSION:
327            msg = f"{self.path} was created by a newer dojinvoice-db (schema v{version} > v{SCHEMA_VERSION})"
328            raise SchemaVersionError(msg)

332    def upsert_work(self, work: Work, *, detail: bool, now: datetime | None = None) -> None:
333        """Insert or update ``work``.
334
335        Args:
336            work: The work to store.
337            detail: Whether ``work`` carries the work-page fields. When false,
338                only the cheap ajax-sourced columns are touched, so existing
339                detail columns are never overwritten with NULLs.
340            now: Timestamp recorded as the fetch time (defaults to now).
341        """
342        stamp = to_iso(now or datetime.now().astimezone())
343        values = _summary_values(work)
344        columns = list(_SUMMARY_COLUMNS)
345        if detail:
346            values |= _detail_values(work)
347            columns += [*_DETAIL_COLUMNS, "detail_fetched_at"]
348            values["detail_fetched_at"] = stamp
349        else:
350            columns.append("summary_fetched_at")
351            values["summary_fetched_at"] = stamp
352        values["product_id"] = work.product_id
353        values["first_seen_at"] = stamp
354        insert_columns = ["product_id", "first_seen_at", *columns]
355        placeholders = ",".join("?" * len(insert_columns))
356        updates = ",".join(f"{c} = excluded.{c}" for c in columns)
357        with self.conn:
358            self.conn.execute(
359                f"insert into work ({','.join(insert_columns)}) values ({placeholders}) "  # noqa: S608
360                f"on conflict (product_id) do update set {updates}",
361                [values[c] for c in insert_columns],
362            )
363            if detail:
364                self._replace_children(work)
365
366    def _replace_children(self, work: Work) -> None:
367        pid = work.product_id
368        creators = [(pid, role, name) for role in CREATOR_ROLES for name in getattr(work, role, None) or []]
369        self.conn.execute("delete from work_creator where product_id = ?", (pid,))
370        self.conn.executemany("insert or ignore into work_creator values (?,?,?)", creators)
371
372        for table, column, items in (
373            ("work_genre", "genre", work.genre),
374            ("work_file_format", "file_format", work.file_format),
375            ("work_event", "event", work.event),
376        ):
377            self.conn.execute(f"delete from {table} where product_id = ?", (pid,))  # noqa: S608
378            self.conn.executemany(
379                f"insert or ignore into {table} (product_id, {column}) values (?,?)",  # noqa: S608
380                [(pid, item) for item in items or []],
381            )
382
383        self.conn.execute("delete from work_sample_image where product_id = ?", (pid,))
384        self.conn.executemany(
385            "insert or ignore into work_sample_image values (?,?,?)",
386            [(pid, i, url) for i, url in enumerate(work.sample_images or [])],
387        )

389    def upsert_stats(self, stats: WorkStats, *, now: datetime | None = None) -> bool:
390        """Store ``stats``, appending to the history only when figures changed.
391
392        Returns:
393            Whether a history row was appended.
394        """
395        stamp = to_iso(now or datetime.now().astimezone())
396        values = {c: getattr(stats, c) for c in _STAT_COLUMNS}
397        previous = self.conn.execute(
398            f"select {','.join(_HISTORY_COLUMNS)} from work_stat where product_id = ?",  # noqa: S608
399            (stats.product_id,),
400        ).fetchone()
401        changed = previous is None or any(previous[c] != values[c] for c in _HISTORY_COLUMNS)
402        columns = ["product_id", *_STAT_COLUMNS, "fetched_at"]
403        placeholders = ",".join("?" * len(columns))
404        updates = ",".join(f"{c} = excluded.{c}" for c in (*_STAT_COLUMNS, "fetched_at"))
405        with self.conn:
406            self.conn.execute(
407                f"insert into work_stat ({','.join(columns)}) values ({placeholders}) "  # noqa: S608
408                f"on conflict (product_id) do update set {updates}",
409                [stats.product_id, *(values[c] for c in _STAT_COLUMNS), stamp],
410            )
411            if changed:
412                self.conn.execute(
413                    f"insert into work_stat_history "  # noqa: S608
414                    f"(product_id, fetched_at, {','.join(_HISTORY_COLUMNS)}) "
415                    f"values ({','.join('?' * (len(_HISTORY_COLUMNS) + 2))})",
416                    [stats.product_id, stamp, *(values[c] for c in _HISTORY_COLUMNS)],
417                )
418        return changed

420    def record_error(self, product_id: str, message: str, *, now: datetime | None = None) -> None:
421        """Record a failed fetch so later runs can skip or retry it."""
422        stamp = to_iso(now or datetime.now().astimezone())
423        with self.conn:
424            self.conn.execute(
425                "insert into fetch_error (product_id, attempts, last_error, last_attempt_at) "
426                "values (?, 1, ?, ?) "
427                "on conflict (product_id) do update set "
428                "attempts = attempts + 1, last_error = excluded.last_error, "
429                "last_attempt_at = excluded.last_attempt_at",
430                (product_id, message[:500], stamp),
431            )
432
433    def clear_error(self, product_id: str) -> None:
434        """Forget a previously recorded fetch failure."""
435        with self.conn:
436            self.conn.execute("delete from fetch_error where product_id = ?", (product_id,))

462    def pending_detail_ids(
463        self,
464        *,
465        limit: int | None = None,
466        stale_before: datetime | None = None,
467        retry_errors: bool = False,
468        now: datetime | None = None,
469    ) -> list[str]:
470        """Return works whose work-page detail should be fetched.
471
472        A work is pending when its detail was never fetched, when it was
473        fetched before ``stale_before``, or when it was fetched while the work
474        was still an announcement and has since been released -- announced
475        works are not re-fetched every run, only once after release.
476        """
477        conditions = [
478            "w.detail_fetched_at is null",
479            (
480                "(w.regist_date is not null and w.detail_fetched_at is not null"
481                " and w.regist_date > w.detail_fetched_at and w.regist_date <= ?)"
482            ),
483        ]
484        params: list[Any] = [to_iso(now or datetime.now().astimezone())]
485        if stale_before is not None:
486            conditions.append("w.detail_fetched_at < ?")
487            params.append(to_iso(stale_before))
488        sql = f"select w.product_id from work w where ({' or '.join(conditions)})"  # noqa: S608
489        if not retry_errors:
490            sql += " and not exists (select 1 from fetch_error e where e.product_id = w.product_id and e.attempts >= ?)"
491            params.append(MAX_ERROR_ATTEMPTS)
492        sql += " order by w.regist_date desc"
493        if limit is not None:
494            sql += " limit ?"
495            params.append(limit)
496        return [row[0] for row in self.conn.execute(sql, params)]

498    def stale_stat_ids(self, *, stale_before: datetime | None = None, limit: int | None = None) -> list[str]:
499        """Return works whose stats are missing or older than ``stale_before``.
500
501        Ordered oldest-first so an interrupted refresh resumes where it left off.
502        """
503        sql = "select w.product_id from work w left join work_stat s on s.product_id = w.product_id"
504        params: list[Any] = []
505        if stale_before is not None:
506            sql += " where s.fetched_at is null or s.fetched_at < ?"
507            params.append(to_iso(stale_before))
508        sql += " order by s.fetched_at is not null, s.fetched_at"
509        if limit is not None:
510            sql += " limit ?"
511            params.append(limit)
512        return [row[0] for row in self.conn.execute(sql, params)]

514    def load_work(self, product_id: str) -> Work | None:
515        """Rebuild a :class:`~dlsite_async.Work` from the stored summary columns.
516
517        Used to fetch a work's detail page on a later run without re-requesting
518        the ajax info that produced the row in the first place.
519        """
520        row = self.conn.execute("select * from work where product_id = ?", (product_id,)).fetchone()
521        if row is None:
522            return None
523        return Work(
524            product_id=row["product_id"],
525            site_id=row["site_id"] or "maniax",
526            maker_id=row["maker_id"],
527            work_name=row["work_name"],
528            age_category=_enum(AgeCategory, row["age_category"]),
529            work_name_masked=row["work_name_masked"],
530            work_type=_enum(WorkType, row["work_type"]),
531            book_type=_enum(BookType, row["book_type"]),
532            work_image=row["work_image"],
533            regist_date=from_iso(row["regist_date"]),
534            title_name_masked=row["series"],
535        )

537    def get_work(self, product_id: str) -> dict[str, Any] | None:
538        """Return one work with its stats and multi-valued fields, or ``None``."""
539        row = self.conn.execute("select * from work where product_id = ?", (product_id,)).fetchone()
540        if row is None:
541            return None
542        result = dict(row)
543        stat = self.conn.execute("select * from work_stat where product_id = ?", (product_id,)).fetchone()
544        result["stats"] = dict(stat) if stat else None
545        creators: dict[str, list[str]] = {}
546        for crow in self.conn.execute(
547            "select role, name from work_creator where product_id = ? order by role, rowid",
548            (product_id,),
549        ):
550            creators.setdefault(crow["role"], []).append(crow["name"])
551        result["creators"] = creators
552        for key, table, column in (
553            ("genres", "work_genre", "genre"),
554            ("file_formats", "work_file_format", "file_format"),
555            ("events", "work_event", "event"),
556        ):
557            result[key] = [
558                r[0]
559                for r in self.conn.execute(
560                    f"select {column} from {table} where product_id = ? order by rowid",  # noqa: S608
561                    (product_id,),
562                )
563            ]
564        result["sample_images"] = [
565            r[0]
566            for r in self.conn.execute(
567                "select url from work_sample_image where product_id = ? order by idx",
568                (product_id,),
569            )
570        ]
571        return result

573    def summary(self) -> dict[str, Any]:
574        """Return aggregate counts for the ``stats`` command."""
575        row = self.conn.execute(
576            "select count(*) as works,"
577            " sum(detail_fetched_at is not null) as detailed,"
578            " min(regist_date) as oldest,"
579            " max(regist_date) as newest"
580            " from work",
581        ).fetchone()
582        return {
583            "works": row["works"],
584            "detailed": row["detailed"] or 0,
585            "pending_detail": row["works"] - (row["detailed"] or 0),
586            "oldest_release": row["oldest"],
587            "newest_release": row["newest"],
588            "circles": self.conn.execute("select count(distinct maker_id) from work").fetchone()[0],
589            "voice_actors": self.conn.execute(
590                "select count(distinct name) from work_creator where role = 'voice_actor'",
591            ).fetchone()[0],
